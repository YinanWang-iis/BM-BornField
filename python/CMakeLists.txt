cmake_minimum_required(VERSION 3.24)
project(bornfield LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(CUDAToolkit REQUIRED)
# becfield_core.cuh + egcnn_generated.hpp (../lammps) and nlohmann_json.hpp (symmetrix libsymmetrix/source)
set(SYMMETRIX_SOURCE "" CACHE PATH "symmetrix libsymmetrix/source directory (for nlohmann_json.hpp)")
pybind11_add_module(_bornfield src/bornfield_ext.cu)
target_include_directories(_bornfield PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../lammps ${SYMMETRIX_SOURCE})
target_compile_options(_bornfield PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
target_link_libraries(_bornfield PRIVATE CUDA::cublas CUDA::cudart)
